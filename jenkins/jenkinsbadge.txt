pipeline {
    agent any 
    stages {
        stage('Build') { 
            steps {
                script {
                    try {
                        printBanner("BUILD STAGE")
                        echo "This is Build stage."

                        // Add a badge for Build stage
                        addBadge(icon: 'gear', text: 'Build Started')
                        
                    } catch (err) {
                        echo "ERROR in Build stage: ${err}"
                        currentBuild.result = 'FAILURE'
                        addBadge(icon: 'error', text: 'Build Failed')
                        error("Failing the pipeline due to Build error.")
                    }
                }
            }
        }
        stage('Test') { 
            steps {
                script {
                    try {
                        printBanner("TEST STAGE")
                        echo "This is Test stage."

                        // Add a badge for Test stage
                        addBadge(icon: 'star-gold', text: 'Test Started')
                        
                    } catch (err) {
                        echo "ERROR in Test stage: ${err}"
                        currentBuild.result = 'FAILURE'
                        addBadge(icon: 'error', text: 'Test Failed')
                        error("Failing the pipeline due to Test error.")
                    }
                }
            }
        }
        stage('Deploy') { 
            steps {
                script {
                    try {
                        printBanner("DEPLOY STAGE")
                        echo "This is Deploy stage."

                        // Add a badge for Deploy stage
                        addBadge(icon: 'package', text: 'Deploy Started')
                        
                    } catch (err) {
                        echo "ERROR in Deploy stage: ${err}"
                        currentBuild.result = 'FAILURE'
                        addBadge(icon: 'error', text: 'Deploy Failed')
                        error("Failing the pipeline due to Deploy error.")
                    }
                }
            }
        }
    }
}

// Function definition
def printBanner(stageName) {
    echo "======================"
    echo "     ${stageName}     "
    echo "======================"
}
